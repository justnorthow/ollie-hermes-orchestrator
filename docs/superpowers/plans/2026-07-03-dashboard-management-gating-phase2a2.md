# Phase 2a.2 — Gated Per-Agent Dashboard Management Surface: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Route the Hermes dashboard **management** surface (skills, cron, config, env, model, profiles, logs, usage, plugins, oauth) through the orchestrator behind an `account_admin+` gate, block the raw browser paths at the nginx edge, make management **per-agent** with a "Managing: [agent]" selector, and hide the management nav for members — closing the last Phase 2 access hole (a member can currently read `/hermes-proxy/api/env` secrets).

**Architecture:** Mirror the Phase 1 sessions treatment. One generic orchestrator proxy `ALL /v1/agents/{agent}/dashboard/{subpath:path}` gates on `authz.admin_denied`, allowlists real dashboard `/api/*` prefixes, and forwards method+body+query to the per-agent Hermes dashboard with the `X-Hermes-Session-Token`. A separate ungated `GET /v1/agents/{agent}/status` stays member-reachable. nginx 403s the raw management subtrees on `/hermes-proxy` and `/dashboard-proxy`. The frontend repoints management calls to a new per-agent `AgentManagementClient`, adds a shared managing-agent selector, and tier-gates the management nav + routes via `useIdentity`. **The orchestrator 403 is the real gate; the frontend hiding is UX (fail-open).**

**Tech Stack:** Orchestrator — Python 3 / FastAPI / httpx / pytest (`TestClient`). Frontend — TypeScript / React / react-router-dom / vitest. nginx config generated by a POSIX `sh` script tested via vitest `execFileSync('sh', …)`.

```python
"""Gated per-agent dashboard MANAGEMENT proxy (agent instantiation Phase 2a.2).

Phase 1 routed sessions through the orchestrator; 2a.2 does the same for the
Hermes dashboard's MANAGEMENT surface (skills, cron, config, env [secrets!],
model, profiles, logs, usage, plugins, oauth). Every management call is gated
account_admin+ via authz.admin_denied and constrained to an allowlist of real
dashboard /api/* subtrees, then forwarded to the per-agent Hermes dashboard with
the session token. A separate, ungated status passthrough stays member-reachable
(chat polls it). The raw /hermes-proxy + /dashboard-proxy management paths are
blocked at the nginx edge (frontend repo) so this gated proxy is the only way in.
"""
import logging

import httpx
from fastapi import APIRouter, Depends, Request
from fastapi.responses import JSONResponse, Response

from src.api import authz
from src.api.sessions import _dashboard_base, _dashboard_headers
from src.auth import require_bearer

_logger = logging.getLogger(__name__)

router = APIRouter(tags=["manage"], dependencies=[Depends(require_bearer)])

# Real Hermes dashboard /api/* subtrees the management surface uses, verified
# against HermesDashboardClient. A subpath is allowed iff it equals one of these
# or starts with "<prefix>/". NOTE the nested ones: usage is analytics/usage,
# plugins is dashboard/plugins, memory providers are dashboard/plugin-providers,
# oauth is providers/oauth — a naive first-segment check would be wrong. Bare
# "dashboard"/"providers" are deliberately NOT allowed (too broad). "sessions"
# and "status" are excluded — they have their own routes.
_ALLOWED_PREFIXES = (
    "skills",
    "cron",
    "config",
    "env",
    "model",
    "profiles",
    "logs",
    "analytics",
    "dashboard/plugins",
    "dashboard/plugin-providers",
    "providers/oauth",
)


def _subpath_allowed(subpath: str) -> bool:
    return any(subpath == p or subpath.startswith(p + "/") for p in _ALLOWED_PREFIXES)


def _forward_headers(request: Request) -> dict:
    # Session token (orchestrator->dashboard auth) + pass through the caller's
    # content-type so PUT/POST JSON bodies reach the dashboard correctly.
    headers = dict(_dashboard_headers())
    ct = request.headers.get("content-type")
    if ct:
        headers["content-type"] = ct
    return headers


@router.get("/v1/agents/{agent}/status")
def agent_status(agent: str, request: Request):
    """Member-reachable dashboard status passthrough (chat polls it). Ungated on
    tier by design — deliberately outside the admin management allowlist."""
    base = _dashboard_base(agent)
    if not base:
        return JSONResponse({"detail": "Dashboard proxy not configured"}, status_code=503)
    try:
        resp = httpx.get(f"{base}/api/status", headers=_dashboard_headers(), timeout=30.0)
    except Exception:
        _logger.warning("agent_status upstream failed", exc_info=True)
        return JSONResponse({"detail": "Dashboard unreachable"}, status_code=502)
    return Response(content=resp.content, status_code=resp.status_code,
                    media_type=resp.headers.get("content-type") or "application/json")


@router.api_route(
    "/v1/agents/{agent}/dashboard/{subpath:path}",
    methods=["GET", "POST", "PUT", "PATCH", "DELETE"],
)
async def manage(agent: str, subpath: str, request: Request):
    # 1. Gate: account_admin+ (identity-less internal callers pass, per the
    #    established trust boundary).
    denied = authz.admin_denied(request)
    if denied:
        return denied
    # 2. Allowlist (real dashboard paths). Non-allowlisted -> 404 (no existence
    #    leak); the "<prefix>/" match also blocks ../ traversal and sessions/status.
    if not _subpath_allowed(subpath):
        return JSONResponse({"detail": "Not found"}, status_code=404)
    # 3. Forward the original METHOD + body + query to the per-agent dashboard.
    base = _dashboard_base(agent)
    if not base:
        return JSONResponse({"detail": "Dashboard proxy not configured"}, status_code=503)
    qs = request.url.query
    url = f"{base}/api/{subpath}" + (f"?{qs}" if qs else "")
    body = await request.body()
    try:
        resp = httpx.request(
            request.method, url, content=body,
            headers=_forward_headers(request), timeout=30.0,
        )
    except Exception:
        _logger.warning("manage upstream failed", exc_info=True)
        return JSONResponse({"detail": "Dashboard unreachable"}, status_code=502)
    # 4. Return upstream verbatim.
    return Response(content=resp.content, status_code=resp.status_code,
                    media_type=resp.headers.get("content-type") or "application/json")
```

```typescript
import type {
  HermesStatus, HermesCronJob, HermesSkill,
  HermesConfigSchema, HermesEnvVar, HermesLogEntry, HermesUsage,
  HermesModelOptionsResponse, HermesProfile, HermesPlugin, HermesMemoryProvider,
} from '../hermes/HermesTypes';

/**
 * Per-agent Hermes dashboard MANAGEMENT client routed through the orchestrator
 * (agent instantiation Phase 2a.2). Every call goes to
 * `/orchestrator-proxy/v1/agents/{agentId}/dashboard/{subpath}` where `subpath`
 * is the REAL dashboard path WITHOUT the `/api/` prefix (the orchestrator adds
 * `/api/`); the orchestrator gates account_admin+ and injects the dashboard
 * session token server-side, so this client sends NO token. `getStatus` targets
 * the separate member-reachable `/v1/agents/{agentId}/status` route.
 *
 * Method names/signatures mirror HermesDashboardClient's management methods so
 * HermesAdapter can use either interchangeably.
 */
export class AgentManagementClient {
  constructor(
    private readonly agentId: string,
    private readonly orchestratorBase: string = '/orchestrator-proxy',
  ) {}

  private dash(subpath: string): string {
    return `${this.orchestratorBase}/v1/agents/${encodeURIComponent(this.agentId)}/dashboard/${subpath}`;
  }

  private async get<T>(subpath: string): Promise<T> {
    const res = await fetch(this.dash(subpath), { headers: {} });
    if (!res.ok) throw new Error(`Management ${res.status}: ${subpath}`);
    const ct = res.headers.get('content-type') ?? '';
    if (ct.includes('text/html')) throw new Error(`Endpoint not available: ${subpath}`);
    return res.json();
  }

  private async post<T>(subpath: string, body?: unknown): Promise<T> {
    const res = await fetch(this.dash(subpath), {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: body !== undefined ? JSON.stringify(body) : undefined,
    });
    if (!res.ok) throw new Error(`Management ${res.status}: ${subpath}`);
    if (res.status === 204 || res.headers.get('content-length') === '0') return undefined as T;
    return res.json();
  }

  private async put<T>(subpath: string, body: unknown): Promise<T> {
    const res = await fetch(this.dash(subpath), {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });
    if (!res.ok) throw new Error(`Management ${res.status}: ${subpath}`);
    if (res.status === 204 || res.headers.get('content-length') === '0') return undefined as T;
    return res.json();
  }

  private async del(subpath: string): Promise<void> {
    const res = await fetch(this.dash(subpath), { method: 'DELETE', headers: {} });
    if (!res.ok) throw new Error(`Management ${res.status}: ${subpath}`);
  }

  // Status — separate member-reachable route, NOT under /dashboard/.
  getStatus(): Promise<HermesStatus> {
    return (async () => {
      const res = await fetch(`${this.orchestratorBase}/v1/agents/${encodeURIComponent(this.agentId)}/status`, { headers: {} });
      if (!res.ok) throw new Error(`Status ${res.status}`);
      return res.json() as Promise<HermesStatus>;
    })();
  }

  getSkills()                              { return this.get<HermesSkill[]>('skills'); }
  getSkill(name: string)                   { return this.get<HermesSkill | null>(`skills/${encodeURIComponent(name)}`); }
  updateSkill(name: string, content: string) {
    return this.put<HermesSkill>(`skills/${encodeURIComponent(name)}`, { content });
  }
  getCronJobs()                            { return this.get<HermesCronJob[]>('cron/jobs'); }
  createCronJob(job: Omit<HermesCronJob, 'id'>) { return this.post<HermesCronJob>('cron/jobs', job); }
  updateCronJob(id: string, patch: Partial<HermesCronJob>) {
    return this.put<HermesCronJob>(`cron/jobs/${id}`, patch);
  }
  deleteCronJob(id: string)                { return this.del(`cron/jobs/${id}`); }
  getConfigSchema()                        { return this.get<HermesConfigSchema>('config/schema'); }
  getConfig()                              { return this.get<Record<string, unknown>>('config'); }
  updateConfig(values: Record<string, unknown>) { return this.put<void>('config', values); }
  getEnv()                                 { return this.get<HermesEnvVar[]>('env'); }
  setEnv(key: string, value: string)       { return this.put<void>('env', { key, value }); }
  deleteEnv(key: string)                   { return this.del(`env/${encodeURIComponent(key)}`); }
  async revealEnv(key: string): Promise<string> {
    const encoded = encodeURIComponent(key);
    for (const [subpath, isReveal] of [[`env/${encoded}/reveal`, true], [`env/${encoded}`, false]] as const) {
      try {
        const r = isReveal
          ? await this.post<{ value: string } | string>(subpath)
          : await this.get<{ key: string; value: string } | string>(subpath);
        if (typeof r === 'string' && r) return r;
        const v = (r as { value?: string })?.value;
        if (typeof v === 'string') return v;
      } catch { /* try next */ }
    }
    throw new Error('Value not available');
  }
  getLogs(opts: { level?: string; component?: string; limit?: number } = {}) {
    const p = new URLSearchParams();
    if (opts.level) p.set('level', opts.level);
    if (opts.component) p.set('component', opts.component);
    if (opts.limit) p.set('lines', String(opts.limit));  // Hermes API uses 'lines', not 'limit'
    return this.get<HermesLogEntry[]>(`logs?${p}`);
  }
  getUsage(period = 'month')               { return this.get<HermesUsage>(`analytics/usage?period=${period}`); }
  getModels()                              { return this.get<HermesModelOptionsResponse>('model/options'); }
  getOAuthProviders()                      { return this.get<unknown>('providers/oauth'); }
  setModel(provider: string, model: string) {
    return this.post<void>('model/set', { scope: 'main', provider, model });
  }
  getProfiles()                            { return this.get<HermesProfile[]>('profiles'); }
  setActiveProfile(id: string)             { return this.post<void>(`profiles/${id}/activate`); }
  getPlugins()                             { return this.get<HermesPlugin[]>('dashboard/plugins'); }
  enablePlugin(name: string)               { return this.post<void>(`dashboard/plugins/${name}/enable`); }
  disablePlugin(name: string)              { return this.post<void>(`dashboard/plugins/${name}/disable`); }
  getMemoryProviders()                     { return this.get<HermesMemoryProvider[]>('dashboard/plugin-providers'); }
  setActiveMemoryProvider(name: string, config?: Record<string, unknown>) {
    return this.put<void>('dashboard/plugin-providers/active', { name, config });
  }
}
```

```typescript
import { createContext, useContext, useMemo, useState, type ReactNode } from 'react';
import { useIdentity } from '../hooks/useIdentity';

interface ManagingAgentValue {
  agentId: string;
  setAgentId: (id: string) => void;
  agentIds: string[];
}

const Ctx = createContext<ManagingAgentValue | null>(null);

/** Holds which agent the admin is currently managing (Phase 2a.2). Options come
 *  from whoami.reachableAgentIds; defaults to the first reachable id (Ollie/default). */
export function ManagingAgentProvider({ children }: { children: ReactNode }) {
  const { reachableAgentIds } = useIdentity();
  const agentIds = reachableAgentIds.length > 0 ? reachableAgentIds : ['default'];
  const [selected, setSelected] = useState<string | null>(null);
  // Effective selection falls back to the first reachable id until the user picks.
  const agentId = selected && agentIds.includes(selected) ? selected : agentIds[0];
  const value = useMemo<ManagingAgentValue>(
    () => ({ agentId, setAgentId: setSelected, agentIds }),
    [agentId, agentIds],
  );
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

export function useManagingAgent(): ManagingAgentValue {
  const v = useContext(Ctx);
  if (!v) throw new Error('useManagingAgent must be used within ManagingAgentProvider');
  return v;
}
```
